# =============================================================================
# CSE644 Cloud Computing - Requirement 7
# HAProxy acting as a reverse proxy / load balancer in front of Nginx.
#
# Backends are addressed as "web1" and "web2" - Docker's embedded DNS on a
# user-defined bridge network resolves those service names to container IPs,
# so no hard-coded addresses are needed.
# =============================================================================

global
    log stdout format raw local0 info
    maxconn 4096
    # NOTE: no `user`/`group` here. The official haproxy image already starts
    # the process as the unprivileged `haproxy` user, so asking HAProxy to
    # drop privileges again would fail (it cannot setuid when not root).

defaults
    mode    http
    log     global
    option  httplog
    # Preserve the real client IP for the backend in X-Forwarded-For.
    option  forwardfor
    # Reconnect rather than erroring if a backend closes an idle connection.
    option  redispatch
    retries 3
    timeout connect 5s
    timeout client  30s
    timeout server  30s
    timeout http-request 10s

# ---------------------------------------------------------------- frontend --
frontend fe_http
    bind *:80

    # Headers that make the proxy hop visible from the client side. Seeing
    # these in a curl response is the proof that the request went THROUGH
    # HAProxy rather than straight to Nginx.
    http-response set-header X-CSE644-Proxy        "haproxy"
    http-response set-header X-CSE644-Proxy-Node   "%[env(HOSTNAME)]"
    http-request  set-header X-CSE644-Forwarded-By "haproxy"

    default_backend be_nginx

# ----------------------------------------------------------------- backend --
backend be_nginx
    balance roundrobin

    # Active health checking: HAProxy polls /healthz on each Nginx backend and
    # removes any server that stops returning 200. This is what makes the
    # failover demonstration in the evidence script work.
    option httpchk
    http-check send meth GET uri /healthz
    http-check expect status 200

    # init-addr last,libc,none
    #   HAProxy resolves server hostnames when it PARSES the config. Inside a
    #   `docker build` there is no Docker DNS, so "web1" cannot resolve and
    #   HAProxy would abort with "Failed to initialize server(s) addr".
    #   This directive says: try the last known address, then libc DNS, and if
    #   both fail, start anyway with the server marked down instead of
    #   refusing to boot. It also protects against HAProxy starting before the
    #   backend containers have DNS entries.
    server web1 web1:80 check inter 2s fall 2 rise 2 init-addr last,libc,none
    server web2 web2:80 check inter 2s fall 2 rise 2 init-addr last,libc,none

# ------------------------------------------------------------ stats page ----
# Read-only dashboard showing backend health and per-server request counts.
frontend fe_stats
    bind *:8404
    stats enable
    stats uri /
    stats refresh 5s
    stats show-legends

--- [2] Nginx backend configuration ---------------------------------
# Nginx backend behind HAProxy (CSE644 Requirement 7).
# SSI is enabled so the page can print the container's own hostname, making
# round-robin load balancing visible in a browser, not just in headers.

server {
    listen 80;
    server_name _;

    root  /usr/share/nginx/html;
    index index.html;

    # Identify which backend answered. HAProxy passes this straight through,
    # so a client sees it change as round-robin rotates.
    add_header X-CSE644-Backend $hostname always;

    location / {
        ssi on;
        try_files $uri $uri/ =404;
    }

    # Polled by HAProxy's `http-check`. Taking this endpoint down (see the
    # failover step in the evidence script) causes HAProxy to mark the
    # backend DOWN and route all traffic to its peer.
    location = /healthz {
        access_log off;
        add_header Content-Type text/plain always;
        return 200 "ok\n";
    }
}

#
#   docker compose -f req07-haproxy-nginx/docker-compose.yml up -d --build
#   curl http://localhost:8090/

#
# Host port 8090 is used because 8080 is occupied by an unrelated container
# on this machine.
# =============================================================================

services:

  # --- Backend #1 ------------------------------------------------------------
  web1:
    build:
      context: ./nginx
      args:
        BUILD_DATE: "build-time"
    image: auth3nticai/cse644-proxy-nginx:v1
    container_name: cse644-web1
    hostname: web1
    networks: [proxy-net]
    # `expose` publishes the port to other containers on the network ONLY.
    # Deliberately NOT `ports:` - if the host could reach Nginx directly, the
    # demo would not prove that traffic flows through HAProxy.
    expose: ["80"]
    restart: unless-stopped

  # --- Backend #2 ------------------------------------------------------------
  web2:
    build:
      context: ./nginx
    image: auth3nticai/cse644-proxy-nginx:v1
    container_name: cse644-web2
    hostname: web2
    networks: [proxy-net]
    expose: ["80"]
    restart: unless-stopped

  # --- The proxy -------------------------------------------------------------
  haproxy:
    build:
      context: ./haproxy
      args:
        BUILD_DATE: "build-time"
    image: auth3nticai/cse644-haproxy:v1
    container_name: cse644-haproxy
    hostname: haproxy
    networks: [proxy-net]
    ports:
      - "8090:80"     # public entry point
      - "8404:8404"   # HAProxy stats dashboard
    depends_on:
      web1: { condition: service_healthy }
      web2: { condition: service_healthy }
    restart: unless-stopped

networks:
  # A user-defined bridge, which gives containers automatic DNS resolution by
  # service name. That is why haproxy.cfg can say `server web1 web1:80`
  # instead of hard-coding an IP address.
  proxy-net:
    name: cse644-proxy-net
    driver: bridge
